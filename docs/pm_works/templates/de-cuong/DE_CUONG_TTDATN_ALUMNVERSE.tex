% ========== 2.1. GIỚI THIỆU ĐỀ TÀI ==========
\subsection{Giới thiệu đề tài}

Cựu sinh viên (alumni) là nguồn lực quan trọng cho sự phát triển của các trường đại học. Theo Obeng-Ofori và Kwarteng (2021) [1] và UniSel Alumni [2], alumni đóng vai trò thiết yếu: role model, career mentor, cung cấp chuyên môn, hỗ trợ phát triển nghề nghiệp, tuyển sinh, gây quỹ và nâng cao uy tín trường. Mạng lưới alumni hiệu quả giúp kết nối sinh viên -- cựu sinh viên và tạo giá trị đóng góp thực sự.

\textbf{Phân tích hệ thống cũ:} Ứng dụng trước (Prior app 2021) thiên về mô hình mạng xã hội (tìm kiếm CSV, CLB/nhóm, follow, group chat, diễn đàn). Các trang alumni tương tự trên mạng xã hội thường chỉ đăng thông tin một chiều, không có hoạt động gắn kết cụ thể; hơn nữa đặc tính của các trang alumni không phải là ứng dụng để sử dụng thường xuyên, hàng ngày. Dựa trên dữ liệu khảo sát các cộng đồng alumni tương tự (nguồn: Proposed Features for AlumnVerse [3]), tỉ lệ engagement rất thấp (Bảng~\ref{tab:engagement}). Cộng đồng lớn thường ``quan sát, ít trao đổi'', không phát huy đủ giá trị cốt lõi của alumni. Vì vậy nhóm quyết định thay đổi hướng tiếp cận.

\textit{Nhận xét:} Các trang định hướng mạng xã hội thường có độ engagement thấp, không phát triển theo hướng người dùng hàng ngày.

\textbf{Thay đổi hướng tiếp cận:} Alumnverse chuyển sang tập trung vào \textbf{giá trị đóng góp cốt lõi} -- đặc biệt \textbf{mentorship} (hướng dẫn nghề nghiệp, matchmaking mentor--mentee), cơ hội việc làm, chia sẻ chuyên môn, đóng góp/gây quỹ, và vinh danh thành tựu -- thay vì mô hình ``like, follow'' thuần túy.

\textbf{Khảo sát người dùng:} Dự án đã thực hiện khảo sát khoảng 100 người (sinh viên và cựu sinh viên HCMUS) về độ phù hợp tính năng và gợi ý cho Alumnverse. Kết quả hỗ trợ hướng phát triển mentorship, forum Q\&A, networking event, cơ hội nghề nghiệp và kênh đóng góp minh bạch.

\textbf{Ý tưởng giải quyết:} Tái cấu trúc hệ thống, chuyển từ kiến trúc microservices sang \textbf{monolithic} (frontend React, backend Spring WebFlux), thiết kế lại cơ sở dữ liệu, triển khai các module: xác thực, diễn đàn, sự kiện, tin nhắn, nội dung (tin tức, việc làm, thành tựu), \textbf{mentorship}, và \textbf{bộ công cụ multi-tenant} cho phép các khoa khác trong HCMUS tạo trang alumni riêng theo subdomain.

\textbf{Ý nghĩa thực tiễn:} Alumnverse hướng tới \textbf{hệ thống chạy thực tế} -- triển khai production, phục vụ người dùng thật (sinh viên, cựu sinh viên HCMUS) -- không chỉ dừng ở đồ án demo. Hệ thống hỗ trợ HCMUS quản lý mạng lưới alumni, tạo kênh kết nối có giá trị, đồng thời cung cấp \textbf{khung tính năng} tái sử dụng cho nhiều khoa.

% ========== 2.2. MỤC TIÊU ĐỀ TÀI ==========
\subsection{Mục tiêu đề tài}

\textbf{Mục tiêu chính:}

\begin{enumerate}
    \item Xây dựng \textbf{hệ thống chạy thực tế} -- triển khai lên môi trường production, phục vụ người dùng thật, không chỉ là đồ án demo deploy tạm thời.
    \item Tái cấu trúc hệ thống sinh viên -- cựu sinh viên theo kiến trúc hiện đại, dễ mở rộng và bảo trì.
    \item Nâng cao trải nghiệm người dùng qua giao diện web responsive (máy tính và trình duyệt trên điện thoại).
    \item Triển khai các tính năng cốt lõi: xác thực, diễn đàn, sự kiện, tin nhắn, quản lý nội dung, tìm kiếm, mentorship.
\end{enumerate}

\textbf{Mục tiêu cụ thể:}

\begin{itemize}
    \item Phân tích và đánh giá hệ thống hiện tại, xác định điểm cần cải thiện.
    \item Thiết kế kiến trúc monolithic mới (frontend SPA + backend API reactive), hỗ trợ multi-tenant.
    \item Triển khai module xác thực (đăng ký, đăng nhập, OTP, quên mật khẩu), xác minh danh tính người dùng (đối với alumni, sinh viên).
    \item Triển khai phân quyền: master admin quản lý toàn bộ hệ thống Alumnverse; vai trò quản lý của khoa (giáo vụ) quản trị trang alumni của từng khoa.
    \item Triển khai diễn đàn (category, topic, post), đặc biệt diễn đàn sự nghiệp alumni.
    \item Triển khai quản lý sự kiện (CRUD, đăng ký tham dự, vé, check-in).
    \item Triển khai tin nhắn (chat riêng, nhóm).
    \item Triển khai quản lý nội dung: tin tức, tài nguyên học tập, việc làm, thành tựu.
    \item Triển khai \textbf{tìm kiếm}: tìm kiếm cựu sinh viên, nội dung diễn đàn, tin tức, việc làm.
    \item Triển khai \textbf{mentorship}: đăng ký mentor, matchmaking, đặt lịch, theo dõi tiến độ, feedback.
    \item Triển khai \textbf{trang thiết lập admin} (multi-tenant): tạo trang khoa mới, cấu hình bố cục/màu sắc, chọn tính năng, preview, launch.
    \item Kiểm thử và đánh giá hệ thống.
\end{itemize}

\textbf{Kiến thức/kỹ năng đạt được:}

\begin{itemize}
    \item \textbf{Kiến thức:} Kiến trúc phần mềm, REST API, reactive programming, quản lý phiên (JWT).
    \item \textbf{Kỹ thuật:} React, Spring Boot WebFlux, PostgreSQL, Docker.
    \item \textbf{Kỹ năng:} Phân tích yêu cầu, thiết kế hệ thống, làm việc nhóm, quản lý thời gian.
\end{itemize}

% ========== 2.3. PHẠM VI CỦA ĐỀ TÀI ==========
\subsection{Phạm vi của đề tài}

\textbf{Đối tượng nghiên cứu/thực hiện:}

\begin{itemize}
    \item Hệ thống Alumnverse (mã nguồn tại repository dự án).
    \item Người dùng: sinh viên, cựu sinh viên, quản trị viên.
    \item Dữ liệu: thông tin người dùng, diễn đàn, sự kiện, tin nhắn, tin tức, việc làm, thành tựu.
\end{itemize}

\textbf{Đặc điểm:}

Multi-tenant: Alumnverse cung cấp bộ công cụ chung (khung tính năng) để các khoa khác trong HCMUS có thể tạo trang alumni riêng theo subdomain (vd: \texttt{cntt.alumnverse.hcmus.edu.vn}).

\textbf{Hình thức triển khai:}

\begin{itemize}
    \item Web application responsive -- chạy trên trình duyệt máy tính và \textbf{trình duyệt trên điện thoại} (không phát triển ứng dụng native riêng).
    \item Phạm vi đơn vị: HCMUS (Khoa CNTT triển khai đầu tiên, các khoa khác khi có nhu cầu).
    \item Các tính năng nâng cao (AI moderation, chatbot phức tạp) có thể nằm ngoài phạm vi ban đầu.
\end{itemize}

% ========== 2.5. KẾT QUẢ THỰC HIỆN DỰ KIẾN CỦA ĐỀ TÀI ==========
\subsection{Kết quả thực hiện dự kiến của đề tài}

\begin{enumerate}
    \item \textbf{Hệ thống Alumnverse chạy thực tế} -- Triển khai production, phục vụ người dùng thật; web responsive (máy tính và trình duyệt trên điện thoại).
    \item \textbf{Mã nguồn hệ thống} -- Repository hoàn chỉnh (frontend + backend), có thể chạy qua Docker Compose.
    \item \textbf{Báo cáo đồ án} -- Mô tả phân tích, thiết kế, triển khai, đánh giá.
    \item \textbf{Tài liệu} -- Hướng dẫn cài đặt, sử dụng, vận hành.
    \item \textbf{Đo lường và metrics} -- Đồ án đề ra các hoạt động có thể đo lường bằng người dùng thực tế: (i) Thu thập thông số người dùng thực (số đăng ký, lượt truy cập, phiên hoạt động); (ii) Định nghĩa metrics cho đồ án (engagement rate, số mentorship match, lượt đăng bài diễn đàn, tham dự sự kiện); (iii) Báo cáo số liệu từ môi trường production với người dùng thật.
\end{enumerate}
